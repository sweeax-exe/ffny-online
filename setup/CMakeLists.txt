cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(DefJamSetup LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "Windows setup first; macOS/Linux installers are to-dos")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(DefJamLauncher WIN32 main.cpp app.manifest)
target_compile_definitions(DefJamLauncher PRIVATE DEFJAM_LAUNCHER_ONLY UNICODE _UNICODE NOMINMAX)
target_link_libraries(DefJamLauncher PRIVATE comctl32 shell32 ole32 uuid bcrypt)
set(SETUP_PAYLOAD "" CACHE FILEPATH "Verified payload ZIP to embed in the setup executable")
if(SETUP_PAYLOAD)
    file(TO_CMAKE_PATH "${SETUP_PAYLOAD}" PAYLOAD_PATH)
    file(SHA256 "${SETUP_PAYLOAD}" PAYLOAD_SHA256)
    configure_file(payload.rc.in payload.rc @ONLY)
    add_executable(DefJamSetup WIN32 main.cpp app.manifest "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
    target_compile_definitions(DefJamSetup PRIVATE UNICODE _UNICODE NOMINMAX PAYLOAD_SHA256=L"${PAYLOAD_SHA256}")
    target_link_libraries(DefJamSetup PRIVATE comctl32 shell32 ole32 uuid bcrypt)
endif()
